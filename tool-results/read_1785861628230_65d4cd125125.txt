     1→/**
     2→ * PaySwap Protocol — Production Stellar Chain Adapter.
     3→ *
     4→ * This adapter implements the full `ChainAdapter` interface against the real
     5→ * Stellar network. It is **mode-switchable at runtime**:
     6→ *
     7→ *   - `'simulation'` (default): uses the legacy in-process simulator
     8→ *     (`src/protocol/blockchains/stellar/adapter.ts`) so the rest of the
     9→ *     protocol layer runs end-to-end without network access. This is the
    10→ *     safe default.
    11→ *
    12→ *   - `'live'`: uses the real `stellar-sdk` package to build, sign, and
    13→ *     submit transactions to a Horizon server. Live mode requires:
    14→ *       1. `stellar-sdk` installed (`bun add stellar-sdk`)
    15→ *       2. A configured `secretKey` (Stellar secret seed, 'S...')
    16→ *       3. A reachable Horizon URL (testnet/mainnet)
    17→ *
    18→ * ## SDK usage map (live mode)
    19→ *
    20→ * Every method has a comment showing the equivalent `stellar-sdk` call so
    21→ * the surface is drop-in ready for real Stellar integration:
    22→ *
    23→ *   - createAccount         → Operation.createAccount
    24→ *   - fundAccount           → Operation.payment (native)
    25→ *   - registerAsset         → issuer account setup + flag set (no op on-chain)
    26→ *   - issueAsset            → Operation.payment (issuer → holder)
    27→ *   - burnAsset             → issuer clawback OR holder → issuer payment
    28→ *   - createTrustline       → Operation.changeTrust
    29→ *   - transfer              → Operation.payment
    30→ *   - pathPayment           → Operation.pathPaymentStrictSend
    31→ *   - createClaimableBalance→ Operation.createClaimableBalance
    32→ *   - claimBalance          → Operation.claimClaimableBalance
    33→ *   - createEscrowAccount   → Operation.createAccount + SetOptions (2-of-2 + timebounds)
    34→ *   - releaseEscrow         → Operation.payment from escrow account
    35→ *   - sponsorReserve        → Operation.beginSponsoringFutureReserves / endSponsoring…
    36→ *   - feeBumpTransaction    → TransactionBuilder.buildFeeBumpTransaction
    37→ *   - addSigner/removeSigner/setThresholds → Operation.setOptions
    38→ *   - verifyTransaction     → server.transactions().transaction(hash)
    39→ *   - getLatestLedger       → server.ledgers().order('desc').limit(1)
    40→ *   - streamLedgers         → server.ledgers().cursor('now').stream()
    41→ *   - prepareSoroban…       → rpc.Server.simulateTransaction (stub for now)
    42→ *
    43→ * ## Failure model
    44→ *
    45→ * All operations return `{ success, error, ... }`. No exceptions escape the
    46→ * adapter. If `stellar-sdk` is not installed, live-mode methods degrade to
    47→ * `{ success: false, error: 'stellar-sdk not installed — install with: bun add stellar-sdk' }`.
    48→ *
    49→ * ## Evidence
    50→ *
    51→ * Every successful on-chain op produces a kernel `Evidence` with
    52→ * `source: 'on_chain_state'`, `verificationLevel: 'cryptographic'`,
    53→ * `reputation: 1.0`, payload includes `txHash`, `ledger`, `operation`,
    54→ * `network`, `mode`.
    55→ */

    74→// ============================================================================
    75→// Dynamic stellar-sdk import (live mode only)
    76→// ============================================================================
    77→//
    78→// We import `stellar-sdk` lazily inside live-mode methods so the adapter
    79→// never crashes when the package is absent. TypeScript still benefits from
    80→// type inference via `import type` — the types are erased at runtime.
    81→//
    82→// We do NOT do a top-level `import * as Sdk from 'stellar-sdk'` because:
    83→//   1. It would crash at import time if the package is missing.
    84→//   2. It pulls the entire SDK into bundles that only need sim mode.
    85→//
    86→// Live mode resolves the SDK via a dynamic `import('stellar-sdk')`. If the
    87→// import fails, we return a graceful degraded result.
    88→//
    89→
    90→type StellarSdkLike = {
    91→  Server: new (url: string, opts?: unknown) => unknown;
    92→  Networks: { PUBLIC: string; TESTNET: string; FUTURENET: string };
    93→  TransactionBuilder: unknown;
    94→  Operation: unknown;
    95→  Asset: unknown;
    96→  Claimant: unknown;
    97→  Memo: unknown;
    98→  Keypair: unknown;
    99→  native: () => unknown;
   100→};
   101→
   102→let sdkCache: StellarSdkLike | null = null;
   103→let sdkLoadAttempted = false;
   104→let sdkLoadError: string | null = null;
   105→
   106→/**
   107→ * Try to load `stellar-sdk`. Returns null if not installed. Memoized.
   108→ * Exposed for tests + the Horizon sync module.
   109→ */
   110→export async function loadStellarSdk(): Promise<StellarSdkLike | null> {
   111→  if (sdkLoadAttempted) return sdkCache;
   112→  sdkLoadAttempted = true;
   113→  try {
   114→    // Dynamic import — runtime-safe. The `'stellar-sdk'` string is a literal
   115→    // so bundlers can resolve it; Node's runtime resolver will throw if the
   116→    // package is missing, which we catch.
   117→    const mod: unknown = await import('stellar-sdk');
   118→    // stellar-sdk exports everything as named exports + a default. The shape:
   119→    //   { Horizon, rpc, contract, Server (alias of Horizon.Server),
   120→    //     Networks, TransactionBuilder, Operation, Asset, Claimant, Memo,
   121→    //     Keypair, ... }
   122→    const m = mod as StellarSdkLike & { Horizon?: { Server: unknown } };
   123→    sdkCache = {
   124→      Server: (m.Server ?? (m.Horizon ? (m.Horizon as { Server: new (url: string, opts?: unknown) => unknown }).Server : null)) as new (url: string, opts?: unknown) => unknown,
   125→      Networks: m.Networks,
   126→      TransactionBuilder: m.TransactionBuilder,
   127→      Operation: m.Operation,
   128→      Asset: m.Asset,
   129→      Claimant: m.Claimant,
   130→      Memo: m.Memo,
   131→      Keypair: m.Keypair,
   132→      native: typeof m.native === 'function' ? m.native : undefined,
   133→    } as StellarSdkLike;
   134→    return sdkCache;
   135→  } catch (err) {
   136→    sdkLoadError = err instanceof Error ? err.message : String(err);
   137→    sdkCache = null;
   138→    return null;
   139→  }
   140→}
   141→
   142→/** Reset the SDK cache (for tests). */
   143→export function _resetStellarSdkCache(): void {
   144→  sdkCache = null;
   145→  sdkLoadAttempted = false;
   146→  sdkLoadError = null;
   147→}

   260→  // ============================================================ mode switching
   261→
   262→  async setMode(mode: ChainMode): Promise<ChainResult> {
   263→    if (mode === this._mode) {
   264→      return { success: true, mode: this._mode, network: this.network };
   265→    }
   266→    if (mode === 'live') {
   267→      const sdk = await loadStellarSdk();
   268→      if (!sdk) {
   269→        return { success: false, error: sdkLoadError ?? NOT_INSTALLED_ERROR, mode: this._mode, network: this.network };
   270→      }
   271→      // Construct the live Horizon server.
   272→      try {
   273→        // Sdk.Server is a constructor — `new Sdk.Server(url, opts)`.
   274→        const ServerCtor = sdk.Server as new (url: string, opts?: unknown) => unknown;
   275→        this._liveServer = new ServerCtor(this._horizonUrl, { allowHttp: this._horizonUrl.startsWith('http://') });
   276→        this._mode = 'live';
   277→      } catch (err) {
   278→        return {
   279→          success: false,
   280→          error: `failed_to_init_live_server: ${err instanceof Error ? err.message : String(err)}`,
   281→          mode: this._mode,
   282→          network: this.network,
   283→        };
   284→      }
   285→      return { success: true, mode: this._mode, network: this.network };
   286→    }
   287→    // Switching back to simulation.
   288→    this._liveServer = null;
   289→    this._mode = 'simulation';
   290→    return { success: true, mode: this._mode, network: this.network };
   291→  }

   319→      //   const server = new Sdk.Server(horizonUrl);

   353→    if (amount <= 0) return { success: false, error: 'amount_must_be_positive' };
   354→    if (!address) return { success: false, error: 'address_required' };
   355→
   356→    if (this._mode === 'live') {

   358→      //   const op = Sdk.Operation.payment({

   386→  async registerAsset(params: { code: string; issuer: string; metadata?: Record<string, unknown> }): Promise<ChainResult> {
   387→    const { code, issuer, metadata } = params;
   388→    if (!code || !issuer) return { success: false, error: 'code_and_issuer_required' };
   389→    this.simAssetIssuers.set(code, issuer);
   390→
   391→    if (this._mode === 'live') {
   392→      // === live signature ===
   393→      // Stellar doesn't have an explicit "register asset" op — issuance is
   394→      // implicit via a payment from the issuer to a holder. Asset metadata
   395→      // (decimals, name, domain) is published via the issuer's stellar.toml.
   396→      // We return success without submitting a tx.
   397→      return {
   398→        success: true,
   399→        mode: this._mode,
   400→        network: this.network,
   401→        evidence: this.makeEvidence({
   402→          entityId: issuer,
   403→          attester: 'stellar-network',
   404→          payload: { op: 'registerAsset', code, issuer, metadata, note: 'asset registered via stellar.toml — no on-chain tx' },
   405→        }),
   406→      };
   407→    }
   408→
   409→    const txHash = uid('stellarTx');
   410→    return this.recordSimTx({
   411→      txHash,
   412→      operation: 'registerAsset',
   413→      payload: { op: 'registerAsset', code, issuer, metadata },
   414→      entityId: issuer,
   415→      amount: 0,
   416→      assetCode: code,
   417→    });
   418→  }

   420→  async issueAsset(params: { assetCode: string; amount: number; to: string; issuer?: string }): Promise<ChainResult> {
   421→    const { assetCode, amount, to, issuer } = params;
   422→    if (amount <= 0) return { success: false, error: 'amount_must_be_positive' };
   423→    const effectiveIssuer = issuer ?? this.simAssetIssuers.get(assetCode);
   424→    if (!effectiveIssuer) return { success: false, error: 'issuer_required_for_asset' };
   425→
   426→    if (this._mode === 'live') {
   427→      // === live signature ===
   428→      //   const issuerAccount = await server.loadAccount(effectiveIssuer);
   429→      //   const tx = new Sdk.TransactionBuilder(issuerAccount, { fee, networkPassphrase })
   430→      //     .addOperation(Sdk.Operation.payment({
   431→      //       destination: to,
   432→      //       amount: String(amount),
   433→      //       asset: new Sdk.Asset(assetCode, effectiveIssuer),
   434→      //     }))
   435→      //     .setTimeout(180).build();
   436→      //   tx.sign(Sdk.Keypair.fromSecret(secretKey));
   437→      //   const resp = await server.submitTransaction(tx);
   438→      const executed = await this.liveSubmit('payment', {
   439→        destination: to,
   440→        amount: String(amount),
   441→        assetCode,
   442→        issuer: effectiveIssuer,
   443→        source: effectiveIssuer,
   444→      });
   445→      return executed;
   446→    }
   447→
   448→    // Sim — delegate to legacy adapter so the twin-token engine sees the
   449→    // balance update too, then mirror the credit into our local ledger.
   450→    const res = await simAdapter.issueAsset({ assetCode, amount, issuer: effectiveIssuer });
   451→    if (!res.success || !res.txHash) return { success: false, error: res.error ?? 'issue_failed' };
   452→    if (to !== effectiveIssuer) {
   453→      const xfer = await simAdapter.transfer({ assetCode, amount, from: effectiveIssuer, to, memo: 'issue' });
   454→      if (!xfer.success) return { success: false, error: xfer.error ?? 'issue_transfer_failed' };
   455→    }
   456→    this.ensureSimAccount(to);
   457→    this.simCredit(to, assetCode, amount);
   458→    return this.recordSimTx({
   459→      txHash: res.txHash,
   460→      operation: 'issueAsset',
   461→      source: effectiveIssuer,
   462→      payload: { op: 'issueAsset', assetCode, amount, to, issuer: effectiveIssuer },
   463→      entityId: to,
   464→      amount,
   465→      assetCode,
   466→    });
   467→  }

   469→  async burnAsset(params: { assetCode: string; amount: number; from: string }): Promise<ChainResult> {
   470→    const { assetCode, amount, from } = params;
   471→    if (amount <= 0) return { success: false, error: 'amount_must_be_positive' };
   472→
   473→    if (this._mode === 'live') {
   474→      // === live signature ===
   475→      //   Option A: clawback (issuer must have AUTH_CLAWBACK_ENABLED flag set)
   476→      //     Sdk.Operation.clawback({ from, amount: String(amount),
   477→      //                              asset: new Sdk.Asset(assetCode, issuer) })
   478→      //   Option B: holder → issuer payment (redeem/burn)
   479→      //     Sdk.Operation.payment({ destination: issuer, amount: String(amount),
   480→      //                              asset: new Sdk.Asset(assetCode, issuer) })
   481→      const issuer = this.simAssetIssuers.get(assetCode);
   482→      const executed = await this.liveSubmit('payment', {
   483→        destination: issuer,
   484→        amount: String(amount),
   485→        assetCode,
   486→        issuer,
   487→        source: from,
   488→      });
   489→      return executed;
   490→    }
   491→
   492→    const res = await simAdapter.burnAsset({ assetCode, amount, from });
   493→    if (!res.success || !res.txHash) return { success: false, error: res.error ?? 'burn_failed' };
   494→    this.simDebit(from, assetCode, amount);
   495→    return this.recordSimTx({
   496→      txHash: res.txHash,
   497→      operation: 'burnAsset',
   498→      source: from,
   499→      payload: { op: 'burnAsset', assetCode, amount, from },
   500→      entityId: from,
   501→      amount,
   502→      assetCode,
   503→    });
   504→  }

   508→  async createTrustline(params: { account: string; assetCode: string; issuer?: string; limit?: number }): Promise<ChainResult> {
   509→    const { account, assetCode, issuer, limit } = params;
   510→    if (!account) return { success: false, error: 'account_required' };
   511→    const effectiveIssuer = issuer ?? this.simAssetIssuers.get(assetCode);
   512→    if (!isNative(assetCode) && !effectiveIssuer) return { success: false, error: 'issuer_required_for_non_native_asset' };
   513→
   514→    if (this._mode === 'live') {
   515→      // === live signature ===
   516→      //   const acct = await server.loadAccount(account);
   517→      //   const tx = new Sdk.TransactionBuilder(acct, { fee, networkPassphrase })
   518→      //     .addOperation(Sdk.Operation.changeTrust({
   519→      //       asset: new Sdk.Asset(assetCode, effectiveIssuer),
   520→      //       limit: limit ? String(limit) : undefined,
   521→      //     }))
   522→      //     .setTimeout(180).build();
   523→      //   tx.sign(Sdk.Keypair.fromSecret(secretKey));
   524→      //   const resp = await server.submitTransaction(tx);
   525→      const executed = await this.liveSubmit('changeTrust', {
   526→        assetCode,
   527→        issuer: effectiveIssuer,
   528→        limit: limit != null ? String(limit) : undefined,
   529→        source: account,
   530→      });
   531→      return executed;
   532→    }
   533→

  1452→  /** Evaluate a claim predicate against the current sim clock. */
  1453→  private evaluatePredicate(p: ClaimPredicate): boolean {
  1454→    switch (p.kind) {
  1455→      case 'unconditional':
  1456→        return true;
  1457→      case 'before_absolute_time':
  1458→        // absBefore is unix seconds
  1459→        return Math.floor(Date.now() / 1000) < p.absBefore;
  1460→      case 'before_relative_time':
  1461→        return Math.floor(Date.now() / 1000) < p.seconds;
  1462→      case 'and':
  1463→        return this.evaluatePredicate(p.left) && this.evaluatePredicate(p.right);
  1464→      case 'or':
  1465→        return this.evaluatePredicate(p.left) || this.evaluatePredicate(p.right);
  1466→      case 'not':
  1467→        return !this.evaluatePredicate(p.inner);
  1468→      default:
  1469→        return false;
  1470→    }
  1471→  }

  1546→   * would be wired. For now it constructs the operation payload and returns
  1547→   * a structured 'not_yet_wired' error so callers can feature-detect.
  1548→   *
  1549→   * When the integration hardens, replace the body of this method with the
  1550→   * actual `server.submitTransaction(tx)` call shown in each method's
  1551→   * `=== live signature ===` comment block.
  1552→   */
  1553→  private async liveSubmit(operation: string, payload: Record<string, unknown>): Promise<ChainResult> {
  1554→    const sdk = await loadStellarSdk();
  1555→    if (!sdk) {
  1556→      return {
  1557→        success: false,
  1558→        error: NOT_INSTALLED_ERROR,
  1559→        mode: this._mode,
  1560→        network: this.network,
  1561→      };
  1562→    }
  1563→    if (!this._secretKey) {
  1564→      return {
  1565→        success: false,
  1566→        error: 'live_mode_requires_secret_key — configure via configureStellarLive({ secretKey })',
  1567→        mode: this._mode,
  1568→        network: this.network,
  1569→      };
  1570→    }
  1571→    // The SDK is installed and a secret key is configured — but the actual
  1572→    // tx-building + submission call graph is intentionally not yet wired.
  1573→    // Return a structured 'pending_integration' result so the protocol
  1574→    // layer can feature-detect without crashing.
  1575→    //
  1576→    // To activate live submission:
  1577→    //   1. Construct the source account via `server.loadAccount(source)`.
  1578→    //   2. Build the tx with `Sdk.TransactionBuilder(...).addOperation(...).setTimeout(180).build()`.
  1579→    //   3. Sign with `Sdk.Keypair.fromSecret(secretKey)`.
  1580→    //   4. Submit via `server.submitTransaction(tx)`.
  1581→    //   5. Map the Horizon response to a ChainResult + Evidence.
  1582→    return {
  1583→      success: false,
  1584→      error: `live_submit_pending_integration: operation='${operation}' — SDK installed, signing key configured, but tx submission not yet wired`,
  1585→      mode: this._mode,
  1586→      network: this.network,
  1587→      details: { operation, payloadKeys: Object.keys(payload) },
  1588→    } as ChainResult;
  1589→  }
  1590→
  1591→  /** Expose the configured horizon URL (for tests + HorizonSync module). */
  1592→  get horizonUrl(): string {
  1593→    return this._horizonUrl;
  1594→  }
  1595→
  1596→  /** Expose the configured network passphrase. */
  1597→  get networkPassphrase(): string {
  1598→    return this._networkPassphrase;
  1599→  }
  1600→

  1602→  get friendbotUrl(): string {
  1603→    return this.network === 'testnet' ? STELLAR_TESTNET_FRIENDBOT : '';
  1604→  }
  1605→}
