   422→/** Create a Runtime instance. */
   423→export function createRuntime(opts: CreateRuntimeOptions = {}): Runtime {
   424→  const clock: RuntimeClock = opts.virtualClock
   425→    ? new VirtualClock(opts.virtualClock)
   426→    : new LiveClock();
   427→  const eventStore: EventStore = opts.eventStore ?? new InMemoryEventStore();
   428→  const intentEngine = new IntentEngine(clock);
   429→  const policyEngine = new DefaultPolicyEngine();
   430→  const pipeline = new Pipeline(clock, intentEngine, eventStore, policyEngine);
   431→  const projectionRunner = new ProjectionRunner();
   432→  projectionRunner.start(eventStore);
   433→
   434→  // Amendment 1 engines — interface-only implementations for M-RT-1.
   435→  const reserveMarketState = new InMemoryReserveMarket();
