   168→        throw new Error(data?.error || 'Simulation failed');

   171→      toast.success(`Scenario "${data.run.scenarioLabel}" executed`, {

   177→      setRunning(false);

   191→            Pick a pre-built scenario and run it through the same kernel pipeline production uses.

   195→          <div className="grid gap-4 sm:grid-cols-2">

   202→                  <SelectValue placeholder="Choose a scenario" />

   251→                    <CheckCircle2 className="h-8 w-8 text-emerald-500" />

   296→            <Card>

   304→              </CardContent>
   305→            </Card>

   314→            </Card>
   315→          </div>
   316→
   317→          {/* Tabs: Timeline / Events / Ledger / Decisions */}

   323→              <TabsTrigger value="decisions">Decisions ({result.decisions.length})</TabsTrigger>
   324→            </TabsList>
   325→
   326→            {/* Timeline */}

   335→                  </CardDescription>
   336→                </CardHeader>
   337→                <CardContent>

   352→                              {step.status}
   353→                            </Badge>

   355→                          <p className="mt-0.5 text-xs text-muted-foreground">{step.description}</p>
   356→                          {step.detail && (

   372→            {/* Events */}

   519→                {/* Twin tokens + amendments */}
