   133→function fmtDate(iso: string | null): string {
   134→  if (!iso) return '—';
   135→  try {
   136→    return new Date(iso).toLocaleDateString('en-US', {
   137→      month: 'short',
   138→      day: 'numeric',
   139→      year: 'numeric',
   140→    });
   141→  } catch {
   142→    return '—';
   143→  }
   144→}

   146→/**
   147→ * Post an action to the admin publish endpoint. Refreshes the page on
   148→ * success so the server-rendered list re-syncs.
   149→ */
   150→async function postAction(
   151→  id: string,
   152→  action: 'approve' | 'reject' | 'review' | 'suspend' | 'reinstate',
   153→  notes?: string,
   154→): Promise<void> {
   155→  const res = await fetch(`/api/extensions/${id}/publish`, {
   156→    method: 'POST',
   157→    headers: { 'Content-Type': 'application/json' },
   158→    body: JSON.stringify({ action, notes: notes ?? undefined }),
   159→  });
   160→  const data = await res.json();
   161→  if (!res.ok) {
   162→    throw new Error(data?.error || 'Action failed');
   163→  }
   164→}

   166→/**
   167→ * Reject dialog: asks for a reason before flipping status to rejected.
   168→ */
   169→function RejectDialog({
   170→  ext,
   171→  onDone,
   172→}: {
   173→  ext: AdminExtension;
   174→  onDone: () => void;
   175→}) {
   176→  const [open, setOpen] = React.useState(false);
   177→  const [notes, setNotes] = React.useState('');
   178→  const [submitting, setSubmitting] = React.useState(false);
   179→
   180→  async function handle() {
   181→    if (!notes.trim()) {
   182→      toast.error('Please provide a rejection reason');
   183→      return;
   184→    }
   185→    setSubmitting(true);
   186→    try {
   187→      await postAction(ext.id, 'reject', notes.trim());
   188→      toast.success('Extension rejected');
   189→      setOpen(false);
   190→      setNotes('');
   191→      onDone();
   192→    } catch (err) {
   193→      toast.error(err instanceof Error ? err.message : 'Failed to reject');
   194→    } finally {
   195→      setSubmitting(false);
   196→    }
   197→  }
   198→
   199→  return (
   200→    <Dialog open={open} onOpenChange={setOpen}>
   201→      <DialogTrigger asChild>
   202→        <Button variant="outline" size="sm" className="h-8 border-rose-500/40 text-rose-600 hover:bg-rose-500/10 dark:text-rose-400">
   203→          <Ban className="mr-1.5 h-3.5 w-3.5" /> Reject
   204→        </Button>
   205→      </DialogTrigger>
   206→      <DialogContent className="sm:max-w-md">
   207→        <DialogHeader>
   208→          <DialogTitle>Reject “{ext.name}”?</DialogTitle>
   209→          <DialogDescription>
   210→            Provide a clear reason — the developer will see this feedback and
   211→            can re-submit after fixing the issues.
   212→          </DialogDescription>
   213→        </DialogHeader>
   214→        <div className="space-y-1.5">
   215→          <Label htmlFor="reject-notes">Reason</Label>
   216→          <Textarea
   217→            id="reject-notes"
   218→            placeholder="e.g. Missing required permissions documentation…"
   219→            value={notes}
   220→            onChange={(e) => setNotes(e.target.value)}
   221→            className="min-h-24"
   222→          />
   223→        </div>
   224→        <DialogFooter>
   225→          <Button
   226→            variant="outline"
   227→            onClick={() => setOpen(false)}
   228→            disabled={submitting}
   229→          >
   230→            Cancel
   231→          </Button>
   232→          <Button
   233→            variant="destructive"
   234→            onClick={handle}
   235→            disabled={submitting}
   236→          >
   237→            {submitting ? (
   238→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   239→            ) : (
   240→              <Ban className="mr-2 h-4 w-4" />
   241→            )}
   242→            Reject extension
   243→          </Button>
   244→        </DialogFooter>
   245→      </DialogContent>
   246→    </Dialog>
   247→  );
   248→}

   502→/**
   503→ * Review-row card with admin actions: approve, put under review, reject,
   504→ * view details. Suspended rows show a reinstate button.
   505→ */
   506→function ReviewCard({
   507→  ext,
   508→  onDone,
   509→}: {
   510→  ext: AdminExtension;
   511→  onDone: () => void;
   512→}) {
   513→  const [busy, setBusy] = React.useState<string | null>(null);
   514→
   515→  async function run(
   516→    action: 'approve' | 'review' | 'reinstate',
   517→    label: string,
   518→  ) {
   519→    setBusy(label);
   520→    try {
   521→      await postAction(ext.id, action);
   522→      toast.success(
   523→        action === 'approve'
   524→          ? 'Extension published'
   525→          : action === 'review'
   526→            ? 'Moved to review'
   527→            : 'Extension reinstated',
   528→      );
   529→      onDone();
   530→    } catch (err) {
   531→      toast.error(err instanceof Error ? err.message : 'Action failed');
   532→    } finally {
   533→      setBusy(null);
   534→    }
   535→  }
   536→
   537→  return (
   538→    <Card className="flex flex-col">
   539→      <CardHeader>
   540→        <div className="flex items-start justify-between gap-3">
   541→          <div className="min-w-0">
   542→            <CardTitle className="truncate text-base">{ext.name}</CardTitle>
   543→            <div className="mt-1 flex flex-wrap items-center gap-2">
   544→              <StatusBadge status={ext.status} />
   545→              <Badge
   546→                variant="outline"
   547→                className="text-[10px] font-medium capitalize"
   548→              >
   549→                {ext.category}
   550→              </Badge>
   551→              <Badge
   552→                variant="outline"
   553→                className="font-mono text-[10px] font-medium"
   554→              >
   555→                v{ext.version}
   556→              </Badge>
   557→            </div>
   558→          </div>
   559→        </div>
   560→      </CardHeader>
   561→      <CardContent className="flex flex-1 flex-col gap-4">
   562→        <CardDescription className="flex-1 text-xs leading-relaxed">
   563→          {ext.description}
   564→        </CardDescription>
   565→
   566→        <div className="rounded-md border bg-card/50 p-2.5 text-xs">
   567→          <div className="flex items-center justify-between">
   568→            <span className="text-muted-foreground">Developer</span>
   569→            <span className="font-medium">{ext.developer.name}</span>
   570→          </div>
   571→          <div className="mt-1 flex items-center justify-between">
   572→            <span className="text-muted-foreground">Email</span>
   573→            <span className="font-mono text-[11px]">{ext.developer.email}</span>
   574→          </div>
   575→          <div className="mt-1 flex items-center justify-between">
   576→            <span className="text-muted-foreground">Submitted</span>
   577→            <span>{fmtDate(ext.submittedAt)}</span>
   578→          </div>
   579→        </div>
   580→
   581→        {ext.permissions.length > 0 && (
   582→          <div className="flex flex-wrap gap-1.5">
   583→            {ext.permissions.map((p) => (
   584→              <Badge
   585→                key={p}
   586→                variant="secondary"
   587→                className="bg-teal-500/10 text-[10px] font-medium text-teal-600 dark:text-teal-400"
   588→              >
   589→                {p.replace(/_/g, ' ')}
   590→              </Badge>
   591→            ))}
   592→          </div>
   593→        )}
   594→
   595→        {ext.reviewNotes && ext.status === 'rejected' && (
   596→          <div className="rounded-md border border-rose-500/30 bg-rose-500/5 p-2.5 text-xs">
   597→            <div className="font-semibold text-rose-600 dark:text-rose-400">
   598→              Rejection reason
   599→            </div>
   600→            <p className="mt-1 text-muted-foreground">{ext.reviewNotes}</p>
   601→          </div>
   602→        )}
   603→        {ext.reviewNotes && ext.status === 'suspended' && (
   604→          <div className="rounded-md border border-rose-500/30 bg-rose-500/5 p-2.5 text-xs">
   605→            <div className="font-semibold text-rose-600 dark:text-rose-400">
   606→              Suspension reason
   607→            </div>
   608→            <p className="mt-1 text-muted-foreground">{ext.reviewNotes}</p>
   609→          </div>
   610→        )}
   611→
   612→        <div className="flex flex-wrap gap-2 border-t pt-3">
   613→          <DetailsDialog ext={ext} />
   614→
   615→          {ext.status === 'submitted' && (
   616→            <ActionButton
   617→              variant="outline"
   618→              onClick={() => run('review', 'review')}
   619→              loading={busy === 'review'}
   620→            >
   621→              <AlertCircle className="mr-1.5 h-3.5 w-3.5" /> Under review
   622→            </ActionButton>
   623→          )}
   624→
   625→          {['submitted', 'review', 'approved'].includes(ext.status) && (
   626→            <ActionButton
   627→              onClick={() => run('approve', 'approve')}
   628→              loading={busy === 'approve'}
   629→              className="bg-emerald-600 text-white hover:bg-emerald-700"
   630→            >
   631→              <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> Approve &amp; publish
   632→            </ActionButton>
   633→          )}
   634→
   635→          {['submitted', 'review'].includes(ext.status) && (
   636→            <RejectDialog ext={ext} onDone={onDone} />
   637→          )}
   638→
   639→          {ext.status === 'suspended' && (
   640→            <ActionButton
   641→              variant="outline"
   642→              onClick={() => run('reinstate', 'reinstate')}
   643→              loading={busy === 'reinstate'}
   644→            >
   645→              <RotateCcw className="mr-1.5 h-3.5 w-3.5" /> Reinstate
   646→            </ActionButton>
   647→          )}
   648→        </div>
   649→      </CardContent>
   650→    </Card>
   651→  );
   652→}
