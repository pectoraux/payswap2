   423→/** Create a Runtime instance. */
   424→export function createRuntime(opts: CreateRuntimeOptions = {}): Runtime {
   425→  const clock: RuntimeClock = opts.virtualClock
   426→    ? new VirtualClock(opts.virtualClock)
   427→    : new LiveClock();
   428→  // R3 FIX: Use PostgresEventStore as the authoritative event store.
   429→  // The database is the source of truth — events are written to Postgres
   430→  // BEFORE the API returns. The in-memory cache is disposable.
   431→  // Falls back to InMemoryEventStore if DATABASE_URL is not set (for tests).
   432→  const eventStore: EventStore = opts.eventStore
   433→    ?? (process.env.DATABASE_URL
   434→      ? new PostgresEventStore()

   911→    registerStage: (stage, handler) => pipeline.register(stage, handler),
   912→    registerIntent: (kind, hooks) => intentEngine.register(kind, hooks),
   913→  };
   914→
   915→  // M-PLATFORM-38: Wire the runtime reference back to the platform engine.
   916→  (platform as unknown as { inputs: { runtime: Runtime } }).inputs.runtime = runtime;
   917→
   918→  return runtime;
   919→}
   920→
   921→/**
   922→ * The default Runtime singleton (live environment, in-memory store).
   923→ *
   924→ * Uses globalThis so Next.js dev-mode module re-instantiation doesn't
   925→ * create duplicate runtimes (same pattern as the existing eventBus/db).
   926→ */
   927→const globalForRuntime = globalThis as unknown as { __PAYSWAP_RUNTIME__?: Runtime };
   928→export const runtime: Runtime =
   929→  globalForRuntime.__PAYSWAP_RUNTIME__ ?? createRuntime();
   930→if (!globalForRuntime.__PAYSWAP_RUNTIME__) {
   931→  globalForRuntime.__PAYSWAP_RUNTIME__ = runtime;
   932→}

   934→/**
   935→ * Convenience: dispatch a raw intent on the default runtime.
   936→ *
   937→ * @example
   938→ *   const result = await dispatch(
   939→ *     { kind: 'payment', raw: { customer: 'Alice', amount: 120, currency: 'USD' } },
   940→ *     { actor: { id: 'usr_1', role: 'merchant' }, environment: 'sandbox', source: 'dashboard' },
   941→ *   );
   942→ */
   943→export function dispatch(
   944→  raw: MerchantIntent,
   945→  ctx: {
   946→    actor: Actor;
   947→    environment: Environment;
   948→    source: IntentSource;
   949→    correlationId?: string;
   950→    causationId?: string;
   951→  },
   952→): Promise<ExecutionResult> {
   953→  return runtime.dispatch(raw, requestContext(ctx));
   954→}
